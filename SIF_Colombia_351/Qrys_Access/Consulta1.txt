SELECT qry_union_Valores_xInmueble.DESCR, qry_union_Valores_xInmueble.Fecha, tbl_Contratos.COD_CTR, Round([qry_union_Valores_xInmueble]![VALOR]*IIf([qry_SumaGLA_xInmueble]![GLA]=0,1,([tbl_Contratos]![GLA]/[qry_SumaGLA_xInmueble]![GLA])),0) AS VALOR
FROM ((qry_union_Valores_xInmueble INNER JOIN tbl_Contratos ON qry_union_Valores_xInmueble.FECHA = tbl_Contratos.FECHA) INNER JOIN qry_SumaGLA_xInmueble ON (tbl_Contratos.COD_INM = qry_SumaGLA_xInmueble.COD_INM) AND (tbl_Contratos.FECHA = qry_SumaGLA_xInmueble.FECHA)) INNER JOIN tbl_Inmuebles ON (qry_union_Valores_xInmueble.Cod_Inm = tbl_Inmuebles.Cod_Inm1) AND (tbl_Inmuebles.Cod_Inm = tbl_Contratos.COD_INM) AND (tbl_Inmuebles.Fecha = tbl_Contratos.FECHA)
GROUP BY qry_union_Valores_xInmueble.DESCR, qry_union_Valores_xInmueble.Fecha, tbl_Contratos.COD_CTR, Round([qry_union_Valores_xInmueble]![VALOR]*IIf([qry_SumaGLA_xInmueble]![GLA]=0,1,([tbl_Contratos]![GLA]/[qry_SumaGLA_xInmueble]![GLA])),0)
HAVING (((Round([qry_union_Valores_xInmueble]![VALOR]*IIf([qry_SumaGLA_xInmueble]![GLA]=0,1,([tbl_Contratos]![GLA]/[qry_SumaGLA_xInmueble]![GLA])),0))<>0));
