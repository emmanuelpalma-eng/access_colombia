SELECT qry_union_Valores_xInmueble.DESCR, qry_union_Valores_xInmueble.Fecha, tbl_Contratos.COD_CTR, Round([qry_union_Valores_xInmueble]![VALOR]*IIf([qry_SumaGLA_xInmueble]![GLA]=0,1,([tbl_Contratos]![GLA]/[qry_SumaGLA_xInmueble]![GLA])),0) AS VALOR
FROM ((qry_union_Valores_xInmueble INNER JOIN tbl_Contratos ON qry_union_Valores_xInmueble.FECHA = tbl_Contratos.FECHA) INNER JOIN qry_SumaGLA_xInmueble ON tbl_Contratos.FECHA = qry_SumaGLA_xInmueble.FECHA) INNER JOIN tbl_Inmuebles ON (qry_union_Valores_xInmueble.Cod_Inm = tbl_Inmuebles.Cod_Inm1) AND (tbl_Contratos.COD_INM = tbl_Inmuebles.Cod_Inm) AND (qry_SumaGLA_xInmueble.COD_INM = tbl_Inmuebles.Cod_Inm1)
GROUP BY qry_union_Valores_xInmueble.DESCR, qry_union_Valores_xInmueble.Fecha, tbl_Contratos.COD_CTR, Round([qry_union_Valores_xInmueble]![VALOR]*IIf([qry_SumaGLA_xInmueble]![GLA]=0,1,([tbl_Contratos]![GLA]/[qry_SumaGLA_xInmueble]![GLA])),0)
HAVING (((qry_union_Valores_xInmueble.Fecha)>#12/31/2013#) AND ((Round([qry_union_Valores_xInmueble]![VALOR]*IIf([qry_SumaGLA_xInmueble]![GLA]=0,1,([tbl_Contratos]![GLA]/[qry_SumaGLA_xInmueble]![GLA])),0))<>0));


UNION 
SELECT "Inversiones en curso (SMArtín hasta Feb-24)" AS DESCR, tbl_EEFF.Fecha, tbl_Cruce_SIF.Cod_SIF AS COD_CTR, Sum(tbl_EEFF.Saldo) AS VALOR
